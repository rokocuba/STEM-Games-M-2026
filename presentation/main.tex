\begin{frame}{Problem koji rješavamo}
    \begin{itemize}
        \item Fokusiramo se na komentare na Redditu: ulaz je samo tekst komentara.
        \item Cilj je binarna klasifikacija: \textbf{HUMAN} ili \textbf{AI}.
        \item Ne koristimo metapodatke: nema korisničke povijesti, karme, vremena objave ni IP adrese.
        \item To čini zadatak realističnim za lokalni alat, ali teži jer model mora prepoznati samo jezične tragove.
    \end{itemize}
\end{frame}

\begin{frame}{Zašto je rast botova važan?}
    \begin{itemize}
        \item Procjenjuje se da velik dio sadržaja na društvenim mrežama više ne pišu ljudi.
        \item Botovi se koriste za spam, phishing, manipulaciju raspravama i umjetno povećanje angažmana.
        \item Chatbotovi mogu proizvoditi uvjerljive komentare u velikoj količini i malom vremenu.
        \item Ako se automatizirani sadržaj ne označi, korisnici teže procjenjuju kome i čemu mogu vjerovati.
    \end{itemize}
\end{frame}

\begin{frame}{Naš detekcijski sustav}
    \begin{itemize}
        \item Sustav kombinira više tekstualnih signala jer jedan indikator nije dovoljno pouzdan.
        \item Perplexity komponenta mjeri koliko je komentar predvidljiv malom jezičnom modelu.
        \item Sentiment komponenta mjeri emocionalni ton i konzistentnost izražavanja.
        \item TF-IDF + logistička regresija u \texttt{tf-logreg/train.py} uči direktno iz riječi, fraza i znakova.
        \item Dodatno ćemo trenirati SVM i druge klasifikatore nad istim ili proširenim značajkama.
        \item Završna neuronska mreža spaja izlaze svih komponenti u jednu odluku HUMAN/AI.
    \end{itemize}
\end{frame}

\begin{frame}{Formalizacija zadatka}
    \begin{itemize}
        \item Svaki komentar pretvaramo u par \((x_i, y_i)\), gdje je \(x_i\) tekst, a \(y_i \in \{0,1\}\).
        \item Koristimo oznaku \(y_i=1\) za AI komentar i \(y_i=0\) za ljudski komentar.
        \item Model uči funkciju:
    \end{itemize}

    \[
        f: \mathcal{X} \rightarrow [0,1], \qquad f(x_i) \approx P(y_i=1 \mid x_i)
    \]

    \begin{itemize}
        \item Konačna odluka dobiva se pragom, najčešće \(0.5\):
    \end{itemize}

    \[
        \hat{y}_i =
        \begin{cases}
            1, & f(x_i) \ge 0.5 \\
            0, & f(x_i) < 0.5
        \end{cases}
    \]
\end{frame}

\begin{frame}{Što radi \texttt{tf-logreg/train.py}?}
    \begin{enumerate}
        \item Učitava CSV s obaveznim stupcima \texttt{comment} i \texttt{label}.
        \item Briše retke bez komentara ili oznake.
        \item Dijeli podatke na trening i test skup uz \texttt{stratify=y}, da omjer klasa ostane sličan.
        \item Gradi \texttt{sklearn} pipeline: ekstrakcija značajki, zatim logistička regresija.
        \item Ispisuje accuracy, classification report i confusion matrix.
        \item Sprema naučeni model u \texttt{tf-logreg/reddit\_ai\_detector.joblib}.
    \end{enumerate}
\end{frame}

\begin{frame}{Značajke: riječi i znakovi}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Word TF-IDF}
            \begin{itemize}
                \item Koristi riječi i parove riječi: \texttt{ngram\_range=(1,2)}.
                \item Hvata fraze, tipične formulacije i tematske obrasce.
                \item \texttt{min\_df=2} izbacuje izraze koji se pojave samo jednom.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Character TF-IDF}
            \begin{itemize}
                \item Koristi nizove znakova duljine 3 do 5.
                \item Hvata interpunkciju, prefikse, sufikse i stil pisanja.
                \item Korisno je za kratke komentare i neobične obrasce teksta.
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{3mm}
    Obje skupine značajki spajaju se pomoću \texttt{FeatureUnion}.
\end{frame}

\begin{frame}{Matematika TF-IDF reprezentacije}
    \begin{block}{Term frequency}
        \[
            \operatorname{tf}(t,d) = \frac{\text{broj pojavljivanja termina } t \text{ u dokumentu } d}{\text{ukupan broj termina u } d}
        \]
    \end{block}

    \begin{block}{Inverse document frequency}
        \[
            \operatorname{idf}(t) = \log \frac{N}{1 + \operatorname{df}(t)}
        \]
    \end{block}

    \begin{block}{TF-IDF težina}
        \[
            x_{t,d} = \operatorname{tf}(t,d) \cdot \operatorname{idf}(t)
        \]
    \end{block}

    \small Rijetki, ali informativni izrazi dobivaju veću težinu; česti i neinformativni izrazi dobivaju manju težinu.
\end{frame}

\begin{frame}{Kakav je to model?}
    \begin{itemize}
        \item Model je \textbf{linearna logistička regresija} nad visoko-dimenzionalnim TF-IDF vektorima.
        \item Ne razumije značenje teksta kao veliki jezični model, nego uči statističke obrasce koji razlikuju klase.
        \item Za svaku značajku nauči težinu: pozitivna težina gura odluku prema jednoj klasi, negativna prema drugoj.
        \item Koristi \texttt{class\_weight="balanced"}, pa ne favorizira veću klasu ako su podaci neuravnoteženi.
        \item Prednost: brz, objašnjiv, radi lokalno na laptopu i može se lako ugraditi u API ili ekstenziju.
    \end{itemize}
\end{frame}

\begin{frame}{Matematička ideja logističke regresije}
    \begin{block}{Linearna odluka}
        \[
            z = w_1 x_1 + w_2 x_2 + \cdots + w_n x_n + b
        \]
    \end{block}

    \begin{block}{Pretvaranje u vjerojatnost}
        \[
            P(\text{AI} \mid x) = \sigma(z) = \frac{1}{1 + e^{-z}}
        \]
    \end{block}

    \begin{itemize}
        \item Ulaz \(x\) je TF-IDF vektor komentara.
        \item Ako je vjerojatnost veća za \texttt{AI}, model vraća \texttt{AI}; inače vraća \texttt{HUMAN}.
        \item \texttt{predict\_proba} daje numeričku sigurnost odluke.
    \end{itemize}
\end{frame}

\begin{frame}{Funkcija gubitka za logističku regresiju}
    \begin{itemize}
        \item Model se trenira tako da minimizira pogrešku između stvarne oznake \(y_i\) i predikcije \(p_i\).
        \item Za binarnu klasifikaciju koristi se log-loss, odnosno binarna unakrsna entropija.
    \end{itemize}

    \[
        p_i = P(y_i=1 \mid x_i) = \sigma(w^T x_i + b)
    \]

    \[
        \mathcal{L}(w,b) = -\frac{1}{m}\sum_{i=1}^{m}
        \left[y_i\log(p_i) + (1-y_i)\log(1-p_i)\right]
    \]

    \begin{itemize}
        \item Ako je model sigurno kriv, gubitak je velik.
        \item Ako je model sigurno točan, gubitak je blizu nule.
    \end{itemize}
\end{frame}

\begin{frame}{Zašto dodati SVM?}
    \begin{itemize}
        \item SVM je jak baseline za tekst jer TF-IDF daje velike, rijetke i gotovo linearno odvojive vektore.
        \item Umjesto direktnog modeliranja vjerojatnosti, SVM traži granicu s najvećom marginom između klasa.
        \item Linearni SVM često dobro radi kada ima puno značajki i relativno malo šuma po značajci.
    \end{itemize}

    \begin{block}{Optimizacijski cilj linearnog SVM-a}
        \[
            \min_{w,b} \frac{1}{2}\lVert w \rVert^2 + C\sum_{i=1}^{m}\max(0, 1-y_i(w^T x_i+b))
        \]
    \end{block}

    \small Ovdje je \(y_i \in \{-1,1\}\), a parametar \(C\) određuje kompromis između široke margine i grešaka na trening skupu.
\end{frame}

\begin{frame}{Više algoritama, različite greške}
    \begin{itemize}
        \item Logistička regresija daje dobro kalibrirane vjerojatnosti i objašnjive težine značajki.
        \item SVM može bolje iskoristiti marginu između klasa u TF-IDF prostoru.
        \item Stabla i boosting modeli mogu koristiti ručno izrađene stilometrijske značajke: duljina rečenica, interpunkcija, ponavljanja, omjer velikih slova.
        \item Perplexity model hvata predvidljivost teksta, što TF-IDF ne mjeri direktno.
        \item Sentiment model hvata emocionalni ton, intenzitet i stabilnost izražavanja.
    \end{itemize}

    \[
        \text{jači sustav} = \text{različiti signali} + \text{različite pogreške} + \text{model za spajanje}
    \]
\end{frame}

\begin{frame}{Spajanje modela neuronskom mrežom}
    \begin{itemize}
        \item Svaki osnovni model vraća score ili vektor značajki.
        \item Te izlaze spajamo u jedan meta-vektor:
    \end{itemize}

    \[
        h = [p_{\text{logreg}},\; s_{\text{svm}},\; p_{\text{perplexity}},\; s_{\text{sentiment}},\; z_{\text{style}}]
    \]

    \begin{itemize}
        \item Neuronska mreža zatim uči nelinearnu funkciju spajanja:
    \end{itemize}

    \[
        a^{(1)} = \operatorname{ReLU}(W_1h + b_1), \qquad
        \hat{p} = \sigma(W_2a^{(1)} + b_2)
    \]

    \begin{itemize}
        \item Time mreža može naučiti da neki modeli imaju veću težinu u određenim situacijama.
    \end{itemize}
\end{frame}

\begin{frame}{Sigurnost predikcije}
    \begin{itemize}
        \item \texttt{predict.py} i lokalni API koriste \texttt{model.predict\_proba}.
        \item Izlaz nije samo oznaka, nego i raspodjela vjerojatnosti po klasama.
        \item Primjer izlaza API-ja: \texttt{label}, \texttt{score}, \texttt{probabilities}, \texttt{reason}.
        \item \texttt{score} je vjerojatnost klase koju je model odabrao.
        \item Visok score znači da je komentar sličan obrascima iz trening podataka; ne znači matematičku sigurnost da je autor zaista bot.
    \end{itemize}
\end{frame}

\begin{frame}{Ograničenja i rizici}
    \begin{itemize}
        \item Model ovisi o kvaliteti i reprezentativnosti označenog skupa podataka.
        \item Ako se stil AI komentara promijeni, stare TF-IDF značajke mogu izgubiti snagu.
        \item Ljudski komentari napisani formalno mogu izgledati kao AI, a AI komentari s namjernim greškama mogu izgledati ljudski.
        \item Vjerojatnosti logističke regresije treba tretirati kao korisnu mjeru pouzdanosti, ali ne kao konačan dokaz.
        \item Zato je najbolje kombinirati ovaj model s perplexity, sentiment i stilometrijskim značajkama.
    \end{itemize}
\end{frame}

\begin{frame}{Daljnji razvoj}
    \begin{itemize}
        \item Trenutni model već radi lokalno i može se koristiti kroz FastAPI endpoint \texttt{/classify}.
        \item Chrome ekstenzija može slati Reddit komentare lokalnom API-ju i prikazivati oznaku korisniku.
        \item Sljedeći korak je meta-model koji prima: TF-IDF score, perplexity score, sentiment značajke i stilometrijske mjere.
        \item Za meta-model su praktični Random Forest, LightGBM ili XGBoost jer dobro rade sa strukturiranim značajkama.
        \item Sustav treba redovito ponovno trenirati jer se botovi i generativni modeli brzo mijenjaju.
    \end{itemize}
\end{frame}

\begin{frame}{Zaključak}
    \begin{itemize}
        \item Bot promet raste i već čini velik dio interneta.
        \item Naš odabrani problem je detekcija AI-generiranih Reddit komentara samo iz teksta.
        \item TF-IDF + logistička regresija daje brz, lokalno izvediv i mjerljiv baseline.
        \item Na test skupu postiže oko 98\% accuracy, ali rezultat treba interpretirati uz ograničenja skupa podataka.
        \item Najjače rješenje je kombinacija više tekstualnih komponenti, a ne oslanjanje na jedan signal.
    \end{itemize}
\end{frame}
